# 1. 背景：这个项目怎么用"加速"

## 1.1 分层

```

        │
        ├─ resolve_acceleration(device, dtype, use_graph, use_flash)
        │     → (backend_name, graph_enabled)
        │
        └─ create_acceleration(...) → AccelInference（门面）
                │
                └─ Accel/PyTorch/AR/backends/

```

`Accel/MLX/` 是 Apple 的整套实现 —— 说明这套抽象本来就是为多后端准备的。

## 1.2 当前在 Intel 上的行为

`tools/acceleration.py:60-66`：

```python
def resolve_acceleration(device, dtype, use_cuda_graph=True, use_flash_attention=True):
```

```python
def flash_attention_available(device, dtype):
    dev = torch.device(device)
    if not _accel_available(dev):
        return False
    if dtype not in (torch.float16, torch.bfloat16):
        return False              # ← fp32 在本机恒走这里
    if dev.type == "xpu":
        return False              # XPU 上 flash_attn 不可用
    if torch.cuda.get_device_capability(dev)[0] < 8:
        return False
    module = _flash_attention_module()
    return callable(getattr(module, "flash_attn_with_kvcache", None))
```

```python
def resolve_acceleration(device, dtype, use_cuda_graph=True, use_flash_attention=True):
    dev = torch.device(device)
    graph = bool(use_cuda_graph and cuda_graph_available(dev))

    if use_flash_attention and flash_attention_available(dev, dtype):
        backend = "sdpa_flash_xpu" if dev.type == "xpu" else "flash_attn_varlen_cuda_graph"

    return None, False
```

## 5.2 新建 `GPT_SoVITS/Accel/graph_backend.py`

```python
"""图捕获后端抽象。CUDAGraph 与 XPUGraph 的 API 差异全收在这里。"""

import contextlib

import torch


class CudaGraphBackend:
    name = "cuda"

    @staticmethod
    def stream():
        return torch.cuda.Stream()

    @staticmethod
    def current_stream():
        return torch.cuda.current_stream()

    @staticmethod
    @contextlib.contextmanager
    def stream_ctx(s):
        with torch.cuda.stream(s):
            yield

    @staticmethod
    def new_graph():
        return torch.cuda.CUDAGraph()

    @staticmethod
    @contextlib.contextmanager
    def graph_ctx(graph, pool=None):
        with torch.cuda.graph(graph, pool):
            yield

    @staticmethod
    def sync():
        torch.cuda.synchronize()


class XpuGraphBackend:
    name = "xpu"

    @staticmethod
    def stream():
        return torch.xpu.Stream()

    @staticmethod
    def current_stream():
        return torch.xpu.current_stream()

    @staticmethod
    @contextlib.contextmanager
    def stream_ctx(s):
        with torch.xpu.stream(s):
            yield

    @staticmethod
    def new_graph():
        return torch.xpu.XPUGraph()

    @staticmethod
    @contextlib.contextmanager
    def graph_ctx(graph, pool=None):
        fn = getattr(torch.xpu, "graph", None)
        if fn is None:
            raise RuntimeError("torch.xpu.graph not available in this build")
        try:
            with fn(graph, pool) if pool is not None else fn(graph):
                yield
        except TypeError:
            with fn(graph):
                yield

    @staticmethod
    def sync():
        torch.xpu.synchronize()


def get_backend(device):
    return XpuGraphBackend if torch.device(device).type == "xpu" else CudaGraphBackend
```

`tools/xpu_caps.py`：

```python
"""Intel 侧能力探测。全部运行时实测。"""

import logging

import torch

_log = logging.getLogger(__name__)
_CACHE: dict[str, bool] = {}


def has_xpu_graph() -> bool:
    if "graph" in _CACHE:
        return _CACHE["graph"]
    ok = False
    if hasattr(torch, "xpu") and torch.xpu.is_available():
        try:
            cls = getattr(torch.xpu, "XPUGraph", None)
            if cls is not None:
                g = cls()
                del g
                ok = True
        except Exception as e:
            _log.info("XPUGraph instantiation failed: %s", e)
    _CACHE["graph"] = ok
    return ok
```

## 5.4 改 `t2s_model_abc.py:486-509`

```python
def capture(self, input_pos, x, x_dec, kv_caches, pool=None, *args, **kwds):
    from GPT_SoVITS.Accel.graph_backend import get_backend

    bk = get_backend(x.device)
    is_xpu = torch.device(x.device).type == "xpu"
    warmup = 10 if is_xpu else 5        # Intel 首次 kernel 编译更慢

    s = bk.stream()
    s.wait_stream(bk.current_stream())
    graph = bk.new_graph()

    with bk.stream_ctx(s):
        for _ in range(warmup):
            self.h(x, input_pos, kv_caches, *args, **kwds)

    bk.current_stream().wait_stream(s)

    with bk.graph_ctx(graph, pool):
        x_dec.copy_(self.h(x, input_pos, kv_caches, *args, **kwds))

    bk.sync()
    return graph
```

# 附录 A：跳过 P3 的影响

| 影响 | 程度 |
|---|---|
| `gpt_streaming`（语义流式） | **完全不可用**（抛 ValueError） |
| 普通推理 | 不受影响 |
| 单句速度 | 相比有图慢 1.5~2x（**估计，需实测**） |
| 批量速度 | 慢 1.2~1.4x（**估计**） |
| WebUI 控件 | 两个开关自动隐藏（`visible=graph_available`），界面不残缺 |
| **训练** | **完全不受影响**（训练脚本不 import Accel） |

**结论**：P3a 必做（功能），P3b 可选（速度）。
